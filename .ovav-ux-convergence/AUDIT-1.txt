# Remove duplicate entries introduced by inherited environment managers.
__ovav_dedupe_path() {
    local entry deduped= old_ifs=$IFS
    IFS=:
    for entry in $PATH; do
        [ -n "$entry" ] || continue
        case ":$deduped:" in
            *:"$entry":*) ;;
            *) deduped="${deduped:+$deduped:}$entry" ;;
        esac
    done
    IFS=$old_ifs
    PATH=$deduped
}
